\section{Introduction}
% ============================================================

\begin{frame}{Motivation Home Bias and Risk Sharing}
  \begin{itemize}
    \item Macroeconomic Risk Sharing
      \begin{itemize}
        \item international diversification and smoothing of income streams
        \item hedging against idiosyncratic income shocks
      \end{itemize}
    \end{itemize}

  \begin{itemize}
    \item  Home Bias
      \begin{itemize}
        \item tendency of investors to hold a disproportionally large share of domestic assets
        \item less internationally diversified portfolios than financial theory would predict
      \end{itemize}
    \end{itemize}

    \vfill

  \begin{block}{They do not need to go hand in hand...}
    Agents could also smooth consumption through borrowing and lending. 
  \end{block}

\end{frame}

\section{Panel-data Regressions}
% ============================================================
\begin{frame}{Panel-data Regressions}
  \begin{block}{Income Risk Sharing Panel Regression}
  \begin{equation}\label{eq:gni_panel}
    \Delta \log GNI_{it} - \Delta \log GNI_t
    =
    \text{constant}
    +
    \kappa
    \left(
    \Delta \log GDP_{it} - \Delta \log GDP_t
    \right)
    +
    \epsilon_{it}
  \end{equation}

  with

  \begin{equation}\label{eq:kappa_gni}
    \kappa
    =
    \kappa_0
    +
    \kappa_1(t - \overline{t})
    +
    \kappa_2(EHB_{it} - \overline{EHB}_t)
  \end{equation}
  \end{block}

  \begin{block}{Consumption Risk Sharing Panel Regression}
  \begin{equation}\label{eq:cons_panel}
    \Delta \log C_{it} - \Delta \log C_t
    =
    \text{constant}
    +
    \eta
    \left(
    \Delta \log GDP_{it} - \Delta \log GDP_t
    \right)
    +
    \epsilon_{it}
  \end{equation}

  with

  \begin{equation}\label{eq:eta_cons}
    \eta
    =
    \eta_0
    +
    \eta_1(t - \overline{t})
    +
    \eta_2(EHB_{it} - \overline{EHB}_t)
  \end{equation}
  \end{block}
\end{frame}
