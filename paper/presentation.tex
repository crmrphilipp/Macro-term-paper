\section{Model}
% ============================================================

\begin{frame}{Theoretical Framework}
  Consider a representative agent who maximises
  \begin{equation}\label{eq:utility}
    \max_{\{c_t, k_{t+1}\}} \;
    \sum_{t=0}^{\infty} \beta^{t}\, u(c_t)
    \quad \text{s.t.} \quad
    c_t + k_{t+1} = f(k_t) + (1-\delta)\,k_t
  \end{equation}
  where $\beta\in(0,1)$ is the discount factor and $\delta$ the depreciation rate.

  \begin{block}{Euler equation}
    \[
      u'(c_t) = \beta\bigl[f'(k_{t+1}) + 1 - \delta\bigr]\,u'(c_{t+1})
    \]
  \end{block}
\end{frame}
