\section{Replication Sørensen et al. 2007}

\begin{frame}{Home Bias - Theory}

\textbf{Concept of Home Bias}
\begin{itemize}
    \item same market portfolio of risky assets based on Capital Asset Pricing Model (CAPM)
    \item countries hold equities/bonds of each country corresponding to each countries share of world market capitalisation
    \item deviations from this standard are considered a ``Home Bias''
\end{itemize}

\vspace{0.3cm}

\textbf{Potential Explanations for Home Bias}
\begin{itemize}
    \item currency risk
    \item transaction costs
\end{itemize}

\end{frame}

\begin{frame}{Home Bias - Measurement}
    \begin{equation}
      \text{Equity Home Bias}_i = 1 - \frac{w_i^{\text{foreign}}}{1 - A_i}
    \end{equation}
    where:
    \begin{equation*}
      w_i^{\text{foreign}} = \frac{\text{Foreign equity held by country } i}{\text{Country } i\text{'s total equity portfolio}}
    \end{equation*}

    \begin{equation*}
      A_i = \frac{\text{Stock market capitalization of country } i}{\text{Stock market capitalization of the world}}
    \end{equation*}

    and

    \begin{equation*}
      \text{Total equity portfolio}_i = \text{Market cap}_i + \text{Foreign equity held}_i - \text{Domestic equity held by foreigners}_i
    \end{equation*}
  \end{frame}

\begin{frame}{Cross-section Risk Sharing Regressions: Reminder}
  \underline{Setup} \\
  Risk sharing regression estimating the \alert{degree of consumption} \tiny{(Lecture 4, slide 24)} \normalsize
    \begin{itemize}
    \item $\Delta \log C_{it} - \Delta \log C_t = \alpha + \beta_{C,t}(\Delta \log GDP_{it} - \Delta \log GDP_t) + \varepsilon_{it}$
  \end{itemize}
  \underline{Perfect risk sharing}\\
  $\alert{\beta_{c,t} = 0} \implies$ Consumption perfectly tracks world consumption ! \\[0.2cm]
  \underline{Exercise \tiny{(Figures 1,2 of reference paper)}} \\
  Plot rolling $\alert{(1-\widehat{\beta_{c,t}})*100}$ against foreign asset holdings over GDP
  \begin{itemize}
    \item Confirmation $\surd$: \alert{Increasing trend} of risk sharing and financial openness
    \item Caveats $\times$: Specific \alert{levels} of risk sharing and overall increase
  \end{itemize}

\begin{frame}{Panel-data Regressions}
  
  \begin{block}{Consumption Risk Sharing Panel Regression}
  \begin{equation}\label{eq:cons_panel}
  \Delta \log C_{it} - \Delta \log C_t
  =
    \text{constant}
  +
    \kappa
  \left(
    \Delta \log GDP_{it} - \Delta \log GDP_t
    \right)
  +
    \epsilon_{it}
  \end{equation}
  
  with
  
  \begin{equation}\label{eq:eta_cons}
  \kappa
  =
    \kappa_0
  +
    \kappa_1(t - \overline{t})
  +
    \kappa_2(EHB_{it} - \overline{EHB}_t)
  \end{equation}
  \end{block}

  Focus of the analysis on:
  \begin{equation}\label{eq:kappa_risk sharing}
    Risk Sharing
    =
    1 - \kappa_0
    - 
    \kappa_1(t - \overline{t}) 
    {\color{red} - \kappa_2(EHB_{it} - \overline{EHB}_t)}
  \end{equation}
  \end{frame}

\begin{frame}{The Puzzle: Euro Area Equity Home Bias}

\textbf{Hypothesis:}
\begin{itemize}
  \item The currency union lowers exchange rate risk and transaction cost
  \item After 1999, Euro Area countries' EHB should decline faster than non-EA countries
  \item EA aggragate decline should be less steep than Euro Area countries' EHB decline
  \item Lower EHB induces stronger within Euro Area Risk sharing
\end{itemize}

\begin{frame}{Onshore Offshore Financial Centers (OOFCs)}

\textbf{Explanation:} \textcite{NBERw32275}: \textbf{The Geography of Capital Allocation in the Euro Area} 

\begin{itemize}
  \item Luxembourg, Ireland, Netherlands host massive fund industries
  \item Funds are owned by worldwide investors 
  \item Standard data (e.g. EWN) looks at asset fund owner and origin - not underlying securities
\end{itemize}

\vspace{0.3cm}

\textbf{\textcite{NBERw32275}} 
\begin{itemize}
  \item Use regulatory data and commercial fund holdings data
  \item ``Unwind'' fund shares
  \item Disentangle owners and underlying securities without the funds host country
  \item Data Available online
\end{itemize}

\end{frame}

\begin{frame}{Conclusion}
  \begin{itemize}
    \item Home Bias and Risk Sharing are \alert{negatively} associated (causal relationship?)
    \item Finding in \textcite{main} \alert{robust} against various sample extensions (Note: \alert{specification sensitivity!})
    \item Erronoues Data quality leads to EHB \alert{downward distortion}
    \item Debt Home Bias declined stronger in Euro Area compared to US
    \item Capital Markets \alert{gradually more important} for risk sharing (after GFC)
    \item Consumption Smoothing remains at \alert{low levels} compared to US
    \item Potential Cause: \alert{Incomplete European Integration}, no Capital Markets/Banking Union or Fiscal Transfers
  \end{itemize}
  
\end{frame}
